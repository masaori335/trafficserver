        inbound.req.X-Pre = "yes";

        inbound.req.X-Ext = "yes";

        inbound.req.X-Mid = "yes";
    }

        inbound.req.X-Suf = "yes";
    }
    if inbound.url.path != "content/" with PRE {
        inbound.req.X-Not-Pre = "yes";
    }
}
