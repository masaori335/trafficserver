    %2:addr = call.pure BI_IP_SERVER @2:4
    %3:addr = call.pure BI_IP_OUTBOUND @2:4
    %4:sview = fmt F0, %2, %3 @2:4
    call.eff BI_SET_CLIENT_RESP_HEADER, %1, %4 @2:4
    ret @2:4
